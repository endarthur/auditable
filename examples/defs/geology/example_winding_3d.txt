# generalized winding number — 3D view

evaluate triangulated surfaces against block models using the [generalized winding number](https://igl.ethz.ch/projects/winding-number/) (Jacobson et al., SIGGRAPH 2013). blocks are rendered with three.js — drag to orbit, scroll to zoom.

select a **mesh** and adjust block resolution, then press **evaluate** to run. blocks are colored by their inside/outside proportion: amber = fully inside, blue = fully outside, with a gradient between.

// mesh generators

function boxMesh(x0, y0, z0, x1, y1, z1) {
  const v = new Float32Array([
    x0,y0,z0, x1,y0,z0, x1,y1,z0, x0,y1,z0,
    x0,y0,z1, x1,y0,z1, x1,y1,z1, x0,y1,z1,
  ]);
  const t = new Uint32Array([
    0,2,1, 0,3,2, 4,5,6, 4,6,7,
    0,1,5, 0,5,4, 2,3,7, 2,7,6,
    0,4,7, 0,7,3, 1,2,6, 1,6,5,
  ]);
  return { vertices: v, triangles: t, name: 'box' };
}

function sphereMesh(cx, cy, cz, r, slices, stacks) {
  const verts = [], tris = [];
  for (let j = 0; j <= stacks; j++) {
    const theta = Math.PI * j / stacks;
    const sinT = Math.sin(theta), cosT = Math.cos(theta);
    for (let i = 0; i <= slices; i++) {
      const phi = 2 * Math.PI * i / slices;
      verts.push(cx + r * sinT * Math.cos(phi), cy + r * sinT * Math.sin(phi), cz + r * cosT);
    }
  }
  for (let j = 0; j < stacks; j++) {
    for (let i = 0; i < slices; i++) {
      const a = j * (slices + 1) + i, b = a + slices + 1;
      tris.push(a, b, a + 1, a + 1, b, b + 1);
    }
  }
  return { vertices: new Float32Array(verts), triangles: new Uint32Array(tris), name: 'sphere' };
}

function planeMesh(x0, y0, x1, y1, z, angle) {
  const rad = angle * Math.PI / 180;
  const cos = Math.cos(rad), sin = Math.sin(rad);
  const cx = (x0 + x1) / 2, cz = z;
  const pts = [[x0,y0,z],[x1,y0,z],[x1,y1,z],[x0,y1,z]];
  const verts = [];
  for (const [px, py, pz] of pts) {
    const dx = px - cx, dz = pz - cz;
    verts.push(cx + dx * cos - dz * sin, py, cz + dx * sin + dz * cos);
  }
  return {
    vertices: new Float32Array(verts),
    triangles: new Uint32Array([0,1,2, 0,2,3]),
    name: 'plane',
  };
}

// %manual
// winding number block model — three.js viewer

const THREE = await load("https://esm.sh/three@0.170.0");
const { OrbitControls } = await load("https://esm.sh/three@0.170.0/examples/jsm/controls/OrbitControls.js");
const { Winding } = await load("./ext/winding/index.js");

// state
const meshNames = ["sphere (r=2)", "box (2×2×2)", "tilted plane (30°)"];
const meshBuilders = [
  () => sphereMesh(0, 0, 0, 2, 24, 12),
  () => boxMesh(-1, -1, -1, 1, 1, 1),
  () => planeMesh(-3, -3, 3, 3, 0, 30),
];
const analyticalVols = [(4 / 3) * Math.PI * 8, 8, 0];

let meshIdx = 0, gridRes = 8, subRes = 2, showWireframe = true, cutaway = false;

// widgets (callback mode — no DAG re-execution)
ui.dropdown("mesh", meshNames, null, { onInput: v => { meshIdx = meshNames.indexOf(v); } });
ui.slider("grid resolution", gridRes, { min: 4, max: 40, step: 1, onInput: v => { gridRes = v; } });
ui.slider("sub-sampling", subRes, { min: 1, max: 8, step: 1, onInput: v => { subRes = v; } });
ui.checkbox("show mesh wireframe", showWireframe, { onInput: v => { showWireframe = v; } });
ui.checkbox("cutaway (hide front half)", cutaway, { onInput: v => { cutaway = v; } });

// evaluate button
const btnEval = document.createElement('button');
btnEval.textContent = 'evaluate';
btnEval.style.cssText = 'margin:6px 0;padding:4px 16px;background:var(--au-action);color:#111;border:none;border-radius:3px;font:12px var(--au-font-mono);cursor:pointer';
ui.display(btnEval);

// status + canvas
const status = document.createElement('div');
status.style.cssText = 'font:12px monospace;color:#888;margin:4px 0';
ui.display(status);

const W = Math.min(560, window.innerWidth - 80);
const H = W;
const canvas = ui.canvas(W, H);

const statsEl = document.createElement('div');
statsEl.style.cssText = 'font:12px monospace;color:#888;margin:4px 0';
ui.display(statsEl);

// three.js scene (persistent)
const renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
renderer.setSize(W, H);
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.setClearColor(0x0a0a0a);

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(45, W / H, 0.1, 100);
const controls = new OrbitControls(camera, canvas);
controls.enableDamping = true;
controls.dampingFactor = 0.05;

scene.add(new THREE.AmbientLight(0x404040, 0.5));
const sun = new THREE.DirectionalLight(0xc89b3c, 1.6);
sun.position.set(4, 8, 3);
scene.add(sun);
const fill = new THREE.DirectionalLight(0x4466aa, 0.3);
fill.position.set(-3, 4, -2);
scene.add(fill);

let running = true;
function animate() {
  if (!running) return;
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
animate();

// persistent worker (created once, reused across evaluations)
const winding = await Winding.create({ worker: true, gpu: true });
let dynamicObjects = [];

async function evaluate() {
  btnEval.disabled = true;
  status.textContent = 'evaluating…';

  // clear previous blocks/wireframe
  for (const obj of dynamicObjects) scene.remove(obj);
  dynamicObjects = [];

  const mesh = meshBuilders[meshIdx]();
  const { vertices, triangles } = mesh;
  const halfExtent = meshIdx === 2 ? 4 : 3;
  const isOpen = meshIdx === 2;

  // camera
  camera.position.set(halfExtent * 2, halfExtent * 1.5, halfExtent * 2);
  controls.target.set(0, 0, 0);

  // axes
  const axes = new THREE.AxesHelper(halfExtent * 0.6);
  scene.add(axes);
  dynamicObjects.push(axes);

  // wireframe
  if (showWireframe) {
    const meshGeo = new THREE.BufferGeometry();
    meshGeo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(vertices), 3));
    meshGeo.setIndex(new THREE.BufferAttribute(new Uint32Array(triangles), 1));
    const wireframe = new THREE.WireframeGeometry(meshGeo);
    const wireMat = new THREE.LineBasicMaterial({ color: 0xc89b3c, opacity: 0.25, transparent: true });
    const wireObj = new THREE.LineSegments(wireframe, wireMat);
    scene.add(wireObj);
    dynamicObjects.push(wireObj);
  }

  winding.setMesh(vertices, triangles);

  const blockSize = (2 * halfExtent) / gridRes;
  const t0 = performance.now();
  const result = await winding.evaluate({
    origin: [-halfExtent, -halfExtent, -halfExtent],
    size: [blockSize, blockSize, blockSize],
    count: [gridRes, gridRes, gridRes],
  }, {
    mode: 'proportion',
    resolution: [subRes, subRes, subRes],
    threshold: isOpen ? 0 : 0.5,
    onProgress: frac => { status.textContent = `evaluating… ${(frac * 100).toFixed(0)}%`; },
  });
  const elapsed = (performance.now() - t0).toFixed(0);
  status.textContent = `evaluated in ${elapsed} ms`;

  // block cubes (InstancedMesh for performance)
  const { proportions } = result;
  const nx = gridRes, ny = gridRes, nz = gridRes;
  const cubeGeo = new THREE.BoxGeometry(blockSize * 0.85, blockSize * 0.85, blockSize * 0.85);
  const amber = new THREE.Color(0xc89b3c);
  const blue = new THREE.Color(0x2244aa);

  // count visible blocks first
  const visible = [];
  for (let k = 0; k < nz; k++) {
    for (let j = 0; j < ny; j++) {
      for (let i = 0; i < nx; i++) {
        const p = proportions[i + j * nx + k * nx * ny];
        if (p < 0.01) continue;
        const cx = -halfExtent + (i + 0.5) * blockSize;
        const cy = -halfExtent + (j + 0.5) * blockSize;
        const cz = -halfExtent + (k + 0.5) * blockSize;
        if (cutaway && cx > 0 && cz > 0) continue;
        visible.push({ cx, cy, cz, p });
      }
    }
  }
  const blockCount = visible.length;

  if (blockCount > 0) {
    const mat = new THREE.MeshLambertMaterial({ transparent: true, opacity: 0.85 });
    const instMesh = new THREE.InstancedMesh(cubeGeo, mat, blockCount);
    const dummy = new THREE.Object3D();
    const tmpColor = new THREE.Color();
    for (let idx = 0; idx < blockCount; idx++) {
      const { cx, cy, cz, p } = visible[idx];
      dummy.position.set(cx, cy, cz);
      dummy.updateMatrix();
      instMesh.setMatrixAt(idx, dummy.matrix);
      tmpColor.lerpColors(blue, amber, p);
      instMesh.setColorAt(idx, tmpColor);
    }
    instMesh.instanceMatrix.needsUpdate = true;
    instMesh.instanceColor.needsUpdate = true;
    scene.add(instMesh);
    dynamicObjects.push(instMesh);
  }

  // stats
  let volSum = 0;
  for (let i = 0; i < proportions.length; i++) volSum += proportions[i];
  const volume = volSum * (blockSize ** 3);

  const exact = analyticalVols[meshIdx];
  const errPct = exact > 0 ? ((volume - exact) / exact * 100).toFixed(1) : '—';
  const errStr = exact > 0 ? ` (exact ${exact.toFixed(2)}, error ${errPct}%)` : '';
  const backend = winding.hasWorker ? (winding.hasGPU ? 'worker+gpu' : 'worker+cpu') : (winding.hasGPU ? 'gpu' : 'cpu');
  statsEl.textContent = `${mesh.name} — ${blockCount}/${nx*ny*nz} blocks, vol ≈ ${volume.toFixed(2)}${errStr}, ${elapsed} ms (${backend})`;

  btnEval.disabled = false;
}

btnEval.onclick = evaluate;
evaluate(); // initial run

invalidation.then(() => {
  running = false;
  renderer.dispose();
  winding.terminate();
});
