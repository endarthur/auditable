# export as app

build **standalone reactive apps** from notebooks. use the save tray → **export app** to produce a self-contained HTML file with no editor, no toolbar — just your reactive cells running.

this example uses `// %bare` and defines its own visual style — a light theme with teal accents, scoped under a `.app` wrapper so it doesn't interfere with the editor during development.

/// code
// generate data from selected distribution
function generate(n, dist, sigma) {
  const data = [];
  for (let i = 0; i < n; i++) {
    if (dist === 'normal') {
      const u1 = Math.random(), u2 = Math.random();
      data.push(Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2) * sigma);
    } else if (dist === 'uniform') {
      data.push((Math.random() - 0.5) * 2 * sigma);
    } else {
      data.push(-Math.log(1 - Math.random()) * sigma);
    }
  }
  return data;
}
const data = generate(samples, distribution, spread);
const mean = data.reduce((s, v) => s + v, 0) / data.length;
const variance = data.reduce((s, v) => s + (v - mean) ** 2, 0) / data.length;
const stddev = Math.sqrt(variance);
const min = Math.min(...data);
const max = Math.max(...data);

/// code
// %outputId hist-container
const nBins = 20;
const lo = min, hi = max;
const binWidth = (hi - lo) / nBins || 1;
const bins = new Array(nBins).fill(0);
for (const v of data) {
  const idx = Math.min(Math.floor((v - lo) / binWidth), nBins - 1);
  bins[idx]++;
}
const maxBin = Math.max(...bins);

const c = ui.canvas(360, 120);
const ctx = c.getContext('2d');
ctx.clearRect(0, 0, c.width, c.height);
const barW = c.width / nBins;
ctx.fillStyle = '#1a8a7d';
for (let i = 0; i < nBins; i++) {
  const h = maxBin > 0 ? (bins[i] / maxBin) * (c.height - 10) : 0;
  ctx.fillRect(i * barW + 1, c.height - h, barW - 2, h);
}
if (showMean) {
  const mx = ((mean - lo) / (hi - lo)) * c.width;
  ctx.strokeStyle = '#e05540';
  ctx.lineWidth = 2;
  ctx.setLineDash([4, 3]);
  ctx.beginPath();
  ctx.moveTo(mx, 0);
  ctx.lineTo(mx, c.height);
  ctx.stroke();
}

### how to export

1. open the **save tray** in the toolbar
2. click **export app**
3. this notebook uses `// %bare` — the checkbox starts unchecked (no base styles)
4. the exported file is fully self-contained: your CSS, widgets, and reactive code

styles are scoped under `.app` so they don't affect the editor during development. when exported bare, the app wrapper provides all the styling.
