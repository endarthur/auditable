# adder + GSLIB — cross-language geostatistics

A resource estimation pipeline in adder, a JS-targeting Python dialect. Data generation, variogram computation, kriging, widgets, plotting, and grade-tonnage — all in adder. Pure adder — `gslib` and `sadpan` are bundled and load on first `import`. Values are native JS values: no FFI, no marshalling.

## parameters

/// adder
import math

nHoles = ui.slider("drill holes", 30, {"min": 10, "max": 80, "step": 1})
noise = ui.slider("noise", 0.3, {"min": 0.0, "max": 1.0, "step": 0.05})
seed = ui.slider("seed", 42, {"min": 1, "max": 999, "step": 1})
modelNames = ["spherical", "exponential", "gaussian"]
modelIdx = ui.dropdown("variogram model", modelNames)
vRange = ui.slider("range", 0.30, {"min": 0.05, "max": 0.60, "step": 0.01})
vSill = ui.slider("sill", 0.8, {"min": 0.1, "max": 3.0, "step": 0.05})
vNugget = ui.slider("nugget", 0.05, {"min": 0.0, "max": 0.5, "step": 0.01})
gridN = ui.slider("grid", 60, {"min": 20, "max": 120, "step": 5})

/// md
## data + variogram

/// adder
import random
random.seed(seed)

def _grade(x, y):
    g1 = 2.5 * math.exp(-((x - 0.3)**2 + (y - 0.6)**2) / 0.02)
    g2 = 1.8 * math.exp(-((x - 0.7)**2 + (y - 0.35)**2) / 0.04)
    return max(0.01, g1 + g2 + 0.2 + noise * (random.random() - 0.5))

samples = []
for i in range(nHoles):
    x, y = random.random(), random.random()
    samples.append([x, y, round(_grade(x, y), 3)])

_grades = [p[2] for p in samples]
_n = len(_grades)
_mean = sum(_grades) / _n
_std = (sum((g - _mean)**2 for g in _grades) / (_n - 1)) ** 0.5
print(f"{_n} samples · mean {_mean:.3f} · std {_std:.3f} · CV {_std/_mean:.2f}")

# experimental variogram (O(n²) all-pairs)
_nlag = 12
_lag_size = 0.06
_bins = [[0, 0.0, 0.0] for _ in range(_nlag)]
for i in range(len(samples)):
    xi, yi, vi = samples[i]
    for j in range(i + 1, len(samples)):
        xj, yj, vj = samples[j]
        h = math.sqrt((xi - xj)**2 + (yi - yj)**2)
        for k in range(_nlag):
            if abs(h - (k + 1) * _lag_size) <= _lag_size / 2:
                _bins[k][0] += 1
                _bins[k][1] += (vi - vj)**2
                _bins[k][2] += h
                break

vario_h, vario_gamma = [], []
for k in range(_nlag):
    cnt, sq, hs = _bins[k]
    if cnt > 0:
        vario_h.append(hs / cnt)
        vario_gamma.append(sq / (2 * cnt))

print(f"variogram: {sum(b[0] for b in _bins)} pairs, {len(vario_h)} lags")

## variogram plot

/// adder
import js

_vc = ui.canvas(460, 220)
_vctx = _vc.getContext("2d")
_vctx.fillStyle = "#1a1a1a"
_vctx.fillRect(0, 0, 460, 220)

_pad = {"l": 50, "r": 20, "t": 15, "b": 35}
_pw = 460 - _pad["l"] - _pad["r"]
_ph = 220 - _pad["t"] - _pad["b"]
_maxH = max(vario_h) * 1.1
_maxG = max(max(vario_gamma), vNugget + vSill) * 1.2

def _sx(h):
    return _pad["l"] + (h / _maxH) * _pw

def _sy(g):
    return _pad["t"] + _ph - (g / _maxG) * _ph

_vctx.strokeStyle = "#555"
_vctx.lineWidth = 1
_vctx.beginPath()
_vctx.moveTo(_pad["l"], _pad["t"])
_vctx.lineTo(_pad["l"], _pad["t"] + _ph)
_vctx.lineTo(_pad["l"] + _pw, _pad["t"] + _ph)
_vctx.stroke()
_vctx.fillStyle = "#888"
_vctx.font = "10px monospace"
_vctx.textAlign = "center"
_vctx.fillText("lag distance", _pad["l"] + _pw / 2, 220 - 5)
_vctx.save()
_vctx.translate(12, _pad["t"] + _ph / 2)
_vctx.rotate(-js.Math.PI / 2)
_vctx.fillText("γ(h)", 0, 0)
_vctx.restore()
for i in range(5):
    _vctx.fillStyle = "#666"
    _vctx.fillText(f"{(_maxH / 4) * i:.2f}", _sx((_maxH / 4) * i), _pad["t"] + _ph + 14)
for i in range(4):
    _vctx.textAlign = "right"
    _vctx.fillText(f"{(_maxG / 3) * i:.2f}", _pad["l"] - 5, _sy((_maxG / 3) * i) + 3)

def _modelFn(h):
    if h == 0:
        return 0
    hr = h / vRange
    if modelIdx == "spherical":
        c = vSill if hr >= 1 else vSill * (1.5 * hr - 0.5 * hr**3)
    elif modelIdx == "exponential":
        c = vSill * (1 - math.exp(-3 * hr))
    else:
        c = vSill * (1 - math.exp(-3 * hr * hr))
    return vNugget + c

_vctx.strokeStyle = "#c89b3c"
_vctx.lineWidth = 2
_vctx.beginPath()
for i in range(101):
    h = (_maxH / 100) * i
    y = _sy(_modelFn(h))
    if i == 0:
        _vctx.moveTo(_sx(h), y)
    else:
        _vctx.lineTo(_sx(h), y)
_vctx.stroke()
_vctx.setLineDash([4, 4])
_vctx.strokeStyle = "#555"
_vctx.lineWidth = 0.5
_vctx.beginPath()
_vctx.moveTo(_pad["l"], _sy(vNugget + vSill))
_vctx.lineTo(_pad["l"] + _pw, _sy(vNugget + vSill))
_vctx.stroke()
_vctx.setLineDash([])

for i in range(len(vario_h)):
    _vctx.beginPath()
    _vctx.arc(_sx(vario_h[i]), _sy(vario_gamma[i]), 4, 0, js.Math.PI * 2)
    _vctx.fillStyle = "#fff"
    _vctx.fill()
    _vctx.strokeStyle = "#000"
    _vctx.lineWidth = 1
    _vctx.stroke()

## kriging

/// adder
import js
import gslib

_t0 = js.performance.now()
kResult = gslib.kb2d({"data": samples, "grid": {"nx": gridN, "ny": gridN, "xsiz": 1/gridN, "ysiz": 1/gridN}, "variogram": {"nugget": vNugget, "structures": [{"type": modelIdx, "contribution": vSill, "range": vRange}]}, "search": {"radius": 2.0, "ndmax": min(nHoles, 24)}})
_elapsed = js.performance.now() - _t0

est = kResult.est
estv = kResult["var"]
print(f"kriged {gridN}×{gridN} in {_elapsed:.1f} ms")

## kriging map

/// adder
import js

showVar = ui.checkbox("show variance", False)

/// adder
import js

_gridData = estv if showVar else est
_cmap = std.magma if showVar else std.viridis

_vmin = float("inf")
_vmax = float("-inf")
for i in range(_gridData.length):
    _v = _gridData[i]
    if _v < -900:
        continue
    if _v < _vmin:
        _vmin = _v
    if _v > _vmax:
        _vmax = _v
_vr = _vmax - _vmin if _vmax - _vmin != 0 else 1

_size = min(500, js.window.innerWidth - 80)
_c = ui.canvas(_size, _size)
_ctx = _c.getContext("2d")
_cellW = _size / gridN
for iy in range(gridN):
    for ix in range(gridN):
        v = _gridData[iy * gridN + ix]
        _ctx.fillStyle = "#222" if v < -900 else _cmap((v - _vmin) / _vr)
        _ctx.fillRect(ix * _cellW, iy * _cellW, _cellW + 0.5, _cellW + 0.5)
for p in samples:
    _ctx.beginPath()
    _ctx.arc(p[0] * _size, p[1] * _size, 3, 0, js.Math.PI * 2)
    _ctx.fillStyle = _cmap((p[2] - _vmin) / _vr)
    _ctx.fill()
    _ctx.lineWidth = 1.5
    _ctx.strokeStyle = "#000"
    _ctx.stroke()
display(f"{'variance' if showVar else modelIdx} · {nHoles} holes")

## grade-tonnage

/// adder
import sadpan as pd

_grades = pd.DataFrame({"grade": [float(est[i]) for i in range(gridN * gridN) if float(est[i]) > -900]})
_block_t = 2.7 / (gridN * gridN)

gt_rows = []
for co_x10 in range(0, 31):
    co = co_x10 * 0.1
    above = _grades.filter(lambda row: row["grade"] >= co)
    if len(above) == 0:
        break
    g_s = above['grade']
    t = len(above) * _block_t
    g = g_s.mean()
    gt_rows.append({"cutoff": co, "blocks": len(above), "tonnes": round(t, 3), "grade": round(g, 3), "metal": round(t * g, 3)})

pd.DataFrame(gt_rows)

**Pure adder.** Every cell is adder (except the bootstrap loader). Widgets (`ui.slider`, `ui.canvas`, `ui.checkbox`), colormaps (`std.viridis`), module loading (`load`), and output (`display`) are all available as adder builtins. Adder calls `kb2d()` directly — `samples` is a native JS Array, no conversion needed.
