# Python + GSLIB — cross-language geostatistics

A resource estimation pipeline driven from Python. Data generation, experimental variography, kriging, and grade-tonnage analysis all happen in Python cells. GSLIB's WASM kriging is called from Python via the `call()` bridge. JS handles only widgets and canvas rendering.

## Python pipeline

One Python cell does everything: generates drill holes, computes the experimental variogram, calls GSLIB's `kb2d` for kriging via `call()`, and builds the grade-tonnage table. `call()` is a built-in async bridge that suspends MicroPython, runs the WASM function in a separate task, and returns the result.

/// python
import math, json

# ---- RNG ----
_s = seed
def _rng():
    global _s
    _s = (_s * 1103515245 + 12345) & 0x7fffffff
    return _s / 0x7fffffff

# ---- generate drill holes ----
def _grade(x, y):
    g1 = 2.5 * math.exp(-((x - 0.3)**2 + (y - 0.6)**2) / 0.02)
    g2 = 1.8 * math.exp(-((x - 0.7)**2 + (y - 0.35)**2) / 0.04)
    return max(0.01, g1 + g2 + 0.2 + noise * (_rng() - 0.5))

_pts = []
for i in range(nHoles):
    x, y = _rng(), _rng()
    _pts.append([x, y, round(_grade(x, y), 3)])

_grades = [p[2] for p in _pts]
_n = len(_grades)
_mean = sum(_grades) / _n
_std = (sum((g - _mean)**2 for g in _grades) / (_n - 1)) ** 0.5

print(f"{_n} samples \u00b7 mean {_mean:.3f} \u00b7 std {_std:.3f} \u00b7 CV {_std/_mean:.2f}")

# ---- experimental variogram (pure Python, O(n\u00b2)) ----
_nlag = 12
_lag_size = 0.06
_bins = [[0, 0.0, 0.0] for _ in range(_nlag)]

for i in range(len(_pts)):
    xi, yi, vi = _pts[i]
    for j in range(i + 1, len(_pts)):
        xj, yj, vj = _pts[j]
        h = math.sqrt((xi - xj)**2 + (yi - yj)**2)
        for k in range(_nlag):
            if abs(h - (k + 1) * _lag_size) <= _lag_size / 2:
                _bins[k][0] += 1
                _bins[k][1] += (vi - vj)**2
                _bins[k][2] += h
                break

vario_h, vario_gamma = [], []
for k in range(_nlag):
    cnt, sq, hs = _bins[k]
    if cnt > 0:
        vario_h.append(round(hs / cnt, 6))
        vario_gamma.append(round(sq / (2 * cnt), 6))

print(f"variogram: {sum(b[0] for b in _bins)} pairs, {len(vario_h)} lags")

# ---- kriging via call() — Python calls WASM through async bridge ----
krig_result = await call("_gslib.kb2d", {
    "data": _pts,
    "grid": {"nx": gridN, "ny": gridN, "xsiz": 1.0 / gridN, "ysiz": 1.0 / gridN},
    "variogram": {"nugget": vNugget, "structures": [{"type": modelIdx, "contribution": vSill, "range": vRange}]},
    "search": {"radius": 2.0, "ndmax": min(nHoles, 24)},
})
est = krig_result["est"]
estv = krig_result["var"]

print(f"kriged {gridN}\u00d7{gridN} grid")

# ---- grade-tonnage curve ----
_valid = [v for v in est if v > -900]
_block_t = 2.7 / (gridN * gridN)

gt_rows = []
for co_x10 in range(0, 31):
    co = co_x10 * 0.1
    above = [v for v in _valid if v >= co]
    if not above:
        break
    t = len(above) * _block_t
    g = sum(above) / len(above)
    gt_rows.append([round(co, 1), len(above), round(t, 3), round(g, 3), round(t * g, 3)])

# ---- export for JS rendering ----
pipeline_json = json.dumps({
    "samples": _pts,
    "vario": {"h": vario_h, "gamma": vario_gamma},
    "est": est,
    "estv": estv,
    "gt": gt_rows,
})

## variogram

/// code
const vc = ui.canvas(460, 220);
const vctx = vc.getContext("2d");
vctx.fillStyle = "#1a1a1a"; vctx.fillRect(0, 0, 460, 220);

const pad = {l: 50, r: 20, t: 15, b: 35};
const pw = 460 - pad.l - pad.r, ph = 220 - pad.t - pad.b;
const maxH = Math.max(...lagDist) * 1.1;
const maxG = Math.max(...lagGamma, vNugget + vSill) * 1.2;
const sx = h => pad.l + (h / maxH) * pw;
const sy = g => pad.t + ph - (g / maxG) * ph;

vctx.strokeStyle = "#555"; vctx.lineWidth = 1;
vctx.beginPath(); vctx.moveTo(pad.l, pad.t); vctx.lineTo(pad.l, pad.t + ph); vctx.lineTo(pad.l + pw, pad.t + ph); vctx.stroke();
vctx.fillStyle = "#888"; vctx.font = "10px monospace"; vctx.textAlign = "center";
vctx.fillText("lag distance", pad.l + pw / 2, 220 - 5);
vctx.save(); vctx.translate(12, pad.t + ph / 2); vctx.rotate(-Math.PI / 2); vctx.fillText("\u03b3(h)", 0, 0); vctx.restore();
for (let i = 0; i <= 4; i++) { vctx.fillStyle = "#666"; vctx.fillText(((maxH/4)*i).toFixed(2), sx((maxH/4)*i), pad.t + ph + 14); }
for (let i = 0; i <= 3; i++) { vctx.textAlign = "right"; vctx.fillText(((maxG/3)*i).toFixed(2), pad.l - 5, sy((maxG/3)*i) + 3); }

const modelFn = (h) => {
  if (h === 0) return 0;
  const hr = h / vRange;
  let c;
  if (modelIdx === "spherical") c = hr >= 1 ? vSill : vSill * (1.5 * hr - 0.5 * hr**3);
  else if (modelIdx === "exponential") c = vSill * (1 - Math.exp(-3 * hr));
  else c = vSill * (1 - Math.exp(-3 * hr * hr));
  return vNugget + c;
};
vctx.strokeStyle = "#c89b3c"; vctx.lineWidth = 2; vctx.beginPath();
for (let i = 0; i <= 100; i++) { const h = (maxH/100)*i; i === 0 ? vctx.moveTo(sx(h), sy(modelFn(h))) : vctx.lineTo(sx(h), sy(modelFn(h))); }
vctx.stroke();
vctx.setLineDash([4,4]); vctx.strokeStyle = "#555"; vctx.lineWidth = 0.5;
vctx.beginPath(); vctx.moveTo(pad.l, sy(vNugget+vSill)); vctx.lineTo(pad.l+pw, sy(vNugget+vSill)); vctx.stroke(); vctx.setLineDash([]);

for (let i = 0; i < lagDist.length; i++) {
  vctx.beginPath(); vctx.arc(sx(lagDist[i]), sy(lagGamma[i]), 4, 0, Math.PI * 2);
  vctx.fillStyle = "#fff"; vctx.fill(); vctx.strokeStyle = "#000"; vctx.lineWidth = 1; vctx.stroke();
}

## kriging map

/// code
const showVar = ui.checkbox("show variance", false);

/// code
const gridData = showVar ? estv_arr : est_arr;
const cmap = showVar ? std.magma : std.viridis;

let vmin = Infinity, vmax = -Infinity;
for (let i = 0; i < gridData.length; i++) {
  if (gridData[i] < -900) continue;
  if (gridData[i] < vmin) vmin = gridData[i];
  if (gridData[i] > vmax) vmax = gridData[i];
}
const vr = vmax - vmin || 1;

const size = Math.min(500, window.innerWidth - 80);
const c = ui.canvas(size, size);
const ctx = c.getContext("2d");
const cellW = size / gridN;
for (let iy = 0; iy < gridN; iy++) {
  for (let ix = 0; ix < gridN; ix++) {
    const v = gridData[iy * gridN + ix];
    ctx.fillStyle = v < -900 ? "#222" : cmap((v - vmin) / vr);
    ctx.fillRect(ix * cellW, iy * cellW, cellW + 0.5, cellW + 0.5);
  }
}
for (const [px, py, pv] of samples) {
  ctx.beginPath(); ctx.arc(px * size, py * size, 3, 0, Math.PI * 2);
  ctx.fillStyle = cmap((pv - vmin) / vr); ctx.fill();
  ctx.lineWidth = 1.5; ctx.strokeStyle = "#000"; ctx.stroke();
}
display(`${showVar ? "variance" : modelIdx} \u00b7 ${nHoles} holes`);

---

**how it works:** `call(name, args)` is a Python built-in that bridges the nested-WASM gap. MicroPython is itself WASM (Asyncify), so it can't call another WASM module directly. `call()` suspends MicroPython via `await`, the JS bridge runs the WASM function in a `setTimeout(0)` macrotask (clean stack), and resumes Python with the JSON result. Arguments and results are marshalled via JSON automatically. TypedArrays (like kriging's Float64Array output) are converted to plain arrays.
