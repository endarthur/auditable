# simulate, drill, estimate

A complete resource estimation validation workflow in adder, a JS-targeting Python dialect. Generate a "true" deposit by unconditional Gaussian simulation, sample it with synthetic drillholes, krige the samples, then compare the estimate to the truth. One JS cell bootstraps the extensions; everything else is adder.

/// code collapsed
// %collapsed
await load("@gcu/plot")
await load("@gcu/adder")
await load("@gcu/sadpan")
const gslib = await load("@atra/gslib")

/// md
## parameters

/// adder
import math
import random

gridN = ui.slider("grid", 50, {"min": 20, "max": 80, "step": 5})
vRange = ui.slider("range", 12, {"min": 3, "max": 30, "step": 1})
vSill = ui.slider("sill", 1.0, {"min": 0.2, "max": 3.0, "step": 0.1})
vNugget = ui.slider("nugget", 0.1, {"min": 0.0, "max": 0.5, "step": 0.05})
nHoles = ui.slider("drill holes", 40, {"min": 10, "max": 100, "step": 5})
simSeed = ui.slider("sim seed", 42, {"min": 1, "max": 999, "step": 1})

/// md
## truth — unconditional simulation

Generate the "true" deposit using GSLIB's `sgsim`. This is what we're trying to estimate.

/// adder
vario = {"nugget": vNugget, "structures": [{"type": "spherical", "contribution": vSill, "range": vRange}]}
search = {"radius": max(vRange * 2, 20), "ndmax": 10, "nodmax": 12}
grid = {"nx": gridN, "ny": gridN, "xsiz": 1, "ysiz": 1}

engine = gslib.sgsim({"grid": grid, "variogram": vario, "search": search})
truth = engine.run(simSeed)
print(f"simulated {gridN}\u00d7{gridN} grid ({len(truth)} nodes)")

/// md
## drilling — sample the truth

Random drillhole locations. Each sample reads the true value at the nearest grid node.

/// adder
random.seed(simSeed + 1000)

samples = []
for i in range(nHoles):
    x = random.random() * gridN
    y = random.random() * gridN
    ix = min(int(x), gridN - 1)
    iy = min(int(y), gridN - 1)
    v = truth[iy * gridN + ix]
    samples.append([x, y, v])

vals = [s[2] for s in samples]
mean = sum(vals) / len(vals)
sd = (sum((v - mean)**2 for v in vals) / (len(vals) - 1)) ** 0.5
print(f"{nHoles} samples \u00b7 mean {mean:.3f} \u00b7 std {sd:.3f}")

/// md
## estimation — krige the samples

/// adder
import js

t0 = js.performance.now()
kResult = gslib.kb2d({
    "data": samples,
    "grid": grid,
    "variogram": vario,
    "search": {"radius": max(vRange * 2, 20), "ndmax": min(nHoles, 24)}
})
est = kResult.est
estv = kResult["var"]
print(f"kriged in {js.performance.now() - t0:.0f} ms")

/// md
## comparison — truth vs estimate

/// adder
import plt

vmin = min(truth)
vmax = max(truth)

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=[800, 400])

ax1.imshow(truth, gridN, gridN, cmap="viridis", vmin=vmin, vmax=vmax)
ax1.set_title("truth")

ax2.imshow(est, gridN, gridN, cmap="viridis", vmin=vmin, vmax=vmax, nodata=-999)
xs = [s[0] for s in samples]
ys = [s[1] for s in samples]
ax2.scatter(xs, ys, c="white", s=4)
ax2.set_title(f"estimate ({nHoles} holes)")

display(fig.show())

/// md
## error statistics

/// adder
import sadpan as pd

# build a dataframe of truth vs estimate, filtering unestimated nodes
t_vals = []
e_vals = []
for i in range(gridN * gridN):
    if est[i] > -900:
        t_vals.append(truth[i])
        e_vals.append(est[i])

df = pd.DataFrame({"truth": t_vals, "est": e_vals})
df["error"] = df["est"] - df["truth"]

err = df["error"]
t_s = df["truth"]
e_s = df["est"]

me = err.mean()
mae = err.abs().mean()
rmse = err.apply(lambda v: v**2).mean() ** 0.5

# correlation: cov(t,e) / (std_t * std_e)
cov = ((t_s - t_s.mean()) * (e_s - e_s.mean())).mean()
corr = cov / (t_s.std() * e_s.std()) if t_s.std() > 0 and e_s.std() > 0 else 0

pd.DataFrame([{
    "estimated": f"{len(df)} / {gridN * gridN}",
    "mean error": round(me, 4),
    "MAE": round(mae, 4),
    "RMSE": round(rmse, 4),
    "correlation": round(corr, 4),
    "cond. bias": round(me / rmse, 3) if rmse > 0 else 0,
}])

/// md
---

**Workflow:** sgsim generates the truth (no conditioning data). Random drillholes sample it. Kriging estimates the field from samples alone. Error stats measure how well the estimate recovers the truth. Try increasing holes or decreasing range to see estimation quality change.
