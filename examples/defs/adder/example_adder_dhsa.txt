# drill hole spacing analysis

Determine optimal drill spacing for resource classification. Generate a reference model by unconditional simulation, decimate to progressively wider spacings, re-estimate at each level, and measure how estimation quality degrades. The spacing-vs-correlation curve maps directly to JORC/NI 43-101 confidence categories.

Written in [adder](https://github.com/endarthur/auditable), a Python dialect that compiles to JavaScript. Adder runs inside [auditable](https://endarthur.github.io/auditable/) — a reactive computational notebook in a single HTML file. The code cells below use Python syntax but execute natively in the browser, with direct access to JS libraries like `gslib` (geostatistics compiled to WebAssembly) and `@gcu/plot` (canvas plotting). No server, no install — just open the HTML file.

/// md
## parameters

/// adder
import math
import random

gridN = ui.slider("grid", 60, {"min": 30, "max": 100, "step": 5})
vRange = ui.slider("range", 15, {"min": 5, "max": 40, "step": 1})
vSill = ui.slider("sill", 1.0, {"min": 0.2, "max": 3.0, "step": 0.1})
vNugget = ui.slider("nugget", 0.1, {"min": 0.0, "max": 0.5, "step": 0.05})
simSeed = ui.slider("seed", 42, {"min": 1, "max": 999, "step": 1})

/// md
## reference model

/// adder
vario = {"nugget": vNugget, "structures": [{"type": "spherical", "contribution": vSill, "range": vRange}]}
grid = {"nx": gridN, "ny": gridN, "xsiz": 1, "ysiz": 1}

engine = gslib.sgsim({"grid": grid, "variogram": vario, "search": {"radius": max(vRange * 2, 20), "ndmax": 12, "nodmax": 12}})
truth = engine.run(simSeed)
print(f"reference model: {gridN}\u00d7{gridN} grid, range={vRange}, sill={vSill}, nugget={vNugget}")

/// md
## spacing analysis

/// adder
import js
import sadpan as pd

spacings = [2, 3, 4, 5, 6, 8, 10, 12, 15, 20, 25, 30]
spacings = [s for s in spacings if s < gridN / 2]

results = []
for spacing in spacings:
    samples = []
    for iy in range(0, gridN, spacing):
        for ix in range(0, gridN, spacing):
            samples.append([ix + 0.5, iy + 0.5, truth[iy * gridN + ix]])

    if len(samples) < 4:
        continue

    est = gslib.kb2d({
        "data": samples, "grid": grid, "variogram": vario,
        "search": {"radius": max(vRange * 2.5, spacing * 2), "ndmax": min(len(samples), 24)}
    }).est

    # build paired series, filtering unestimated nodes
    t_vals = [truth[i] for i in range(gridN * gridN) if est[i] > -900]
    e_vals = [est[i] for i in range(gridN * gridN) if est[i] > -900]
    if len(t_vals) < 2:
        continue

    t_s = pd.Series(t_vals)
    e_s = pd.Series(e_vals)
    err = e_s - t_s

    cov = ((t_s - t_s.mean()) * (e_s - e_s.mean())).mean()
    vt = ((t_s - t_s.mean()) * (t_s - t_s.mean())).mean()
    corr = cov / (t_s.std() * e_s.std()) if t_s.std() > 0 and e_s.std() > 0 else 0
    slope = cov / vt if vt > 0 else 0
    rmse = (err.apply(lambda v: v**2).mean()) ** 0.5

    results.append({
        "spacing": spacing, "nsamples": len(samples),
        "corr": round(corr, 4), "slope": round(slope, 4), "rmse": round(rmse, 4),
        "est": est, "samples": samples
    })

results_df = pd.DataFrame([{"spacing": r["spacing"], "nsamples": r["nsamples"], "corr": r["corr"], "slope": r["slope"], "rmse": r["rmse"]} for r in results])
results_df

/// md
## maps — truth vs selected spacings

/// adder
import plt
import js
import math

nPick = min(len(results), 5)
if nPick >= 2:
    show_idx = [int(round(i * (len(results) - 1) / (nPick - 1))) for i in range(nPick)]
else:
    show_idx = list(range(len(results)))
seen = []
for idx in show_idx:
    if idx not in seen:
        seen.append(idx)
show = [results[i] for i in seen]

nPanels = len(show) + 1
nCols = math.ceil(nPanels / 2)
vmin = min(truth)
vmax = max(truth)

mapW = min(220, int((js.window.innerWidth - 40) / nCols))
fig, axes = plt.subplots(2, nCols, figsize=[nCols * mapW, 2 * (mapW + 20)])

axes[0].imshow(truth, gridN, gridN, cmap="viridis", vmin=vmin, vmax=vmax)
axes[0].set_title("truth")

for k in range(len(show)):
    r = show[k]
    axes[k + 1].imshow(r["est"], gridN, gridN, cmap="viridis", vmin=vmin, vmax=vmax, nodata=-999)
    sx = [s[0] for s in r["samples"]]
    sy = [s[1] for s in r["samples"]]
    axes[k + 1].scatter(sx, sy, c="white", s=2)
    axes[k + 1].set_title(f"s={r['spacing']} r={r['corr']:.2f}")

display(fig.show())

## classification curve

/// adder
import plt

sp = list(results_df['spacing'].values)
corr = list(results_df['corr'].values)

fig2, ax2 = plt.subplots(1, 1, figsize=[500, 280])
ax2.plot(sp, corr, color="#c89b3c", linewidth=2, marker="o")
ax2.axhline(0.95, color="#4caf50", linestyle="--")
ax2.axhline(0.85, color="#ff9800", linestyle="--")
ax2.axhline(0.70, color="#f44336", linestyle="--")
ax2.text(sp[0], 0.96, "measured (r=0.95)", color="#4caf50")
ax2.text(sp[0], 0.86, "indicated (r=0.85)", color="#ff9800")
ax2.text(sp[0], 0.71, "inferred (r=0.70)", color="#f44336")
ax2.set_xlabel("drill hole spacing")
ax2.set_ylabel("correlation")
ax2.grid(True)
display(fig2.show())

## slope of regression

Conditional bias. Slope = 1.0 is unbiased; < 1.0 means over-smoothing.

/// adder
import plt

sp = list(results_df['spacing'].values)
slopes = list(results_df['slope'].values)

fig3, ax3 = plt.subplots(1, 1, figsize=[500, 200])
ax3.plot(sp, slopes, color="#c89b3c", linewidth=2, marker="o")

ax3.text(sp[0], 1.01, "unbiased (slope=1)", color="#4caf50")

ax3.set_ylabel("slope")
ax3.grid(True)
display(fig3.show())

**Workflow:** sgsim generates a reference model. At each spacing level, samples are placed on a regular grid and the field is re-estimated by kriging. The maps show how spatial resolution degrades. The classification curve shows which spacing achieves measured, indicated, or inferred confidence. Try increasing range (smoother, wider spacing works) or adding nugget (noisier, need denser drilling).
