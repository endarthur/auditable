const project = plan.parsePlan(samplePlan);
ui.display(`"${project.title}" — ${project.tasks.length} tasks, ${project.templates.length} template(s)`);

/// md
## inspect the project

/// code collapsed
const calInfo = [];
if (project.calendarPreset) calInfo.push(`Preset: ${project.calendarPreset}`);
calInfo.push(`${project.calendar.holidays.length} holidays`);
calInfo.push(`${project.calendar.blocked.length} blocked period(s)`);
ui.display(calInfo.join(" · "));

if (project.templates.length) {
  ui.display("### Templates");
  ui.table(project.templates.map(t => ({
    Name: t.name,
    Tasks: t.tasks.filter(tt => tt.id).length,
    IDs: t.tasks.filter(tt => tt.id).map(tt => tt.id).join(", "),
  })));
}

const linked = project.tasks.filter(t => plan.isLinkedPlanTask(t));
const detached = project.tasks.filter(t => t.id && !plan.isLinkedPlanTask(t));
ui.display(`Linked: ${linked.length} · Independent: ${detached.length}`);

/// md
## schedule + gantt

/// code collapsed
const tasks = plan.buildSchedulerTasks(project.tasks);
const result = plan.schedule(tasks, project.calendar, project.projectStart);

ui.display(`Duration: ${result.projectDuration} days · Critical path: ${result.criticalPath.length} tasks`);

ui.table(result.scheduled.map(t => ({
  ID: t.id, Name: t.name || t.id, Group: t.group || "",
  Start: t.earlyStart.toISOString().slice(0,10),
  Finish: t.earlyFinish.toISOString().slice(0,10),
  Float: t.totalFloat, CP: t.isCritical ? "✱" : "",
})), { sortable: true, filter: true });

/// code collapsed
ui.html(plan.gantt(result, { width: 840, showFloat: true, calendar: project.calendar }));

/// md
## Monte Carlo

/// code collapsed
const mc = plan.monteCarlo(tasks, project.calendar, project.projectStart, {
  iterations: 5000, seed: 42, sensitivity: true,
});
ui.display(`P50: ${mc.projectEnd.p50.toISOString().slice(0,10)}  P75: ${mc.projectEnd.p75.toISOString().slice(0,10)}  P90: ${mc.projectEnd.p90.toISOString().slice(0,10)}`);
ui.html(plan.monteCarloPlot(mc, { width: 840 }));

/// md
## deadline risk

/// code collapsed
if (project.deadlines.length) {
  const deadlineObjs = project.deadlines.filter(d => d.date).map(d => ({
    label: d.label || d.taskId || "Project",
    taskId: d.taskId || null,
    deadline: d.date,
  }));
  ui.html(plan.deadlineRiskPlot(mc, deadlineObjs, { width: 840 }));
} else {
  ui.display("No deadlines defined in this project.");
}

/// md
## sensitivity

/// code collapsed
if (mc.sensitivity) {
  ui.html(plan.tornadoPlot(mc.sensitivity, { width: 840, maxBars: 15 }));
}

/// md
## template instancing

Instance from the first template, then serialize. Change the prefix or mode
and re-run to see different results.

/// code
const prefix = ui.textInput("prefix", "c", { placeholder: "prefix" });
const instMode = ui.dropdown("mode", ["linked", "detached"]);

/// code collapsed
if (project.templates.length) {
  const tpl = project.templates[0];
  const group = prefix.toUpperCase() + "/" + tpl.name;
  const existing = new Set(project.tasks.map(t => t.id));
  const collision = tpl.tasks.some(tt => tt.id && existing.has(prefix + "_" + tt.id));

  if (collision) {
    ui.display(`Prefix "${prefix}" collides with existing task IDs.`);
  } else {
    const newTasks = plan.instancePlanTemplate(tpl, prefix, group, instMode);
    ui.display(`Would add ${newTasks.length} ${instMode} tasks from "${tpl.name}":`);
    ui.table(newTasks.map(t => ({
      ID: t.id, Name: t.name, Group: t.group,
      PERT: `${t.o}/${t.m}/${t.p}`,
      Depends: t.depends,
      Linked: plan.isLinkedPlanTask(t) ? "●" : "",
    })));
  }
}
