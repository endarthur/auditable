# drill hole spacing analysis

Determine optimal drill spacing for resource classification. Generate a reference model by unconditional simulation, decimate to progressively wider spacings, re-estimate at each level, and plot the spacing-vs-correlation classification curve.

/// code
const gridN = ui.slider("grid", 60, {min: 30, max: 100, step: 5});
const vRange = ui.slider("range", 15, {min: 5, max: 40, step: 1});
const vSill = ui.slider("sill", 1.0, {min: 0.2, max: 3.0, step: 0.1});
const vNugget = ui.slider("nugget", 0.1, {min: 0.0, max: 0.5, step: 0.05});
const simSeed = ui.slider("seed", 42, {min: 1, max: 999, step: 1});

/// code collapsed
// %collapsed
const plot = await load("@gcu/plot");
const { sgsim, kb2d } = await load("@atra/gslib");

const vario = { nugget: vNugget, structures: [{ type: "spherical", contribution: vSill, range: vRange }] };
const grid = { nx: gridN, ny: gridN, xsiz: 1, ysiz: 1 };
const searchRadius = Math.max(vRange * 2, 20);

const engine = sgsim({ grid, variogram: vario, search: { radius: searchRadius, ndmax: 12, nodmax: 12 } });
invalidation.then(() => engine.dispose());
const truth = engine.run(simSeed);
display(`reference: ${gridN}×${gridN}, range=${vRange}, sill=${vSill}, nugget=${vNugget}`);

/// md
## spacing analysis

/// code
const spacings = [2, 3, 4, 5, 6, 8, 10, 12, 15, 20, 25, 30].filter(s => s < gridN / 2);
const results = [];

for (const spacing of spacings) {
  const samples = [];
  for (let iy = 0; iy < gridN; iy += spacing)
    for (let ix = 0; ix < gridN; ix += spacing)
      samples.push([ix + 0.5, iy + 0.5, truth[iy * gridN + ix]]);
  if (samples.length < 4) continue;

  const est = kb2d({
    data: samples, grid, variogram: vario,
    search: { radius: Math.max(vRange * 2.5, spacing * 2), ndmax: Math.min(samples.length, 24) },
  }).est;

  let n = 0, st = 0, se = 0, st2 = 0, se2 = 0, ste = 0, serr2 = 0;
  for (let i = 0; i < gridN * gridN; i++) {
    if (est[i] <= -900) continue;
    const t = truth[i], e = est[i];
    n++; st += t; se += e; st2 += t * t; se2 += e * e; ste += t * e; serr2 += (e - t) ** 2;
  }
  if (n < 2) continue;

  const mt = st / n, me = se / n, vt = st2 / n - mt * mt, ve = se2 / n - me * me;
  const cov = ste / n - mt * me;
  results.push({
    spacing, nsamples: samples.length, est, samples,
    corr: vt > 0 && ve > 0 ? cov / Math.sqrt(vt * ve) : 0,
    slope: vt > 0 ? cov / vt : 0,
    rmse: Math.sqrt(serr2 / n),
  });
}

display(`\n${"spacing".padStart(7)}  ${"samples".padStart(7)}  ${"corr".padStart(7)}  ${"slope".padStart(7)}  ${"RMSE".padStart(7)}`);
display(`${"-------".padStart(7)}  ${"-------".padStart(7)}  ${"------".padStart(7)}  ${"------".padStart(7)}  ${"------".padStart(7)}`);
for (const r of results)
  display(`${String(r.spacing).padStart(7)}  ${String(r.nsamples).padStart(7)}  ${r.corr.toFixed(4).padStart(7)}  ${r.slope.toFixed(4).padStart(7)}  ${r.rmse.toFixed(4).padStart(7)}`);

/// md
## maps — truth vs selected spacings

/// code
const nPick = Math.min(results.length, 5);
const showIdx = nPick >= 2
  ? Array.from({length: nPick}, (_, i) => Math.round(i * (results.length - 1) / (nPick - 1)))
  : results.map((_, i) => i);
const show = [...new Set(showIdx)].map(i => results[i]);
const nPanels = show.length + 1;
const nCols = Math.ceil(nPanels / 2);

let vmin = Infinity, vmax = -Infinity;
for (let i = 0; i < truth.length; i++) { if (truth[i] < vmin) vmin = truth[i]; if (truth[i] > vmax) vmax = truth[i]; }

const mapW = Math.min(220, Math.floor((window.innerWidth - 40) / nCols));
const {fig, axes} = plot.subplots(2, nCols, {figsize: [nCols * mapW, 2 * (mapW + 20)]});
// subplots(R, C) returns a nested 2D array when both >1 (matplotlib
// shape); use axes.flat for row-major flat iteration.
const cells = axes.flat;

cells[0].imshow(truth, gridN, gridN, {cmap: 'viridis', vmin, vmax});
cells[0].set_title('truth');

for (let k = 0; k < show.length; k++) {
  const r = show[k];
  cells[k + 1].imshow(r.est, gridN, gridN, {cmap: 'viridis', vmin, vmax, nodata: -999});
  cells[k + 1].scatter(r.samples.map(s => s[0]), r.samples.map(s => s[1]), {c: 'white', s: 2});
  cells[k + 1].set_title(`s=${r.spacing} r=${r.corr.toFixed(2)}`);
}
display(fig.show());

/// md
## classification curve

/// code
const sp = results.map(r => r.spacing);
const corr = results.map(r => r.corr);

const {fig: f2, ax: a2} = plot.subplots(1, 1, {figsize: [500, 280]});
a2.plot(sp, corr, {color: '#c89b3c', linewidth: 2, marker: 'o'});
a2.axhline(0.95, {color: '#4caf50', linestyle: '--'});
a2.axhline(0.85, {color: '#ff9800', linestyle: '--'});
a2.axhline(0.70, {color: '#f44336', linestyle: '--'});
a2.text(sp[0], 0.96, 'measured (r=0.95)', {color: '#4caf50'});
a2.text(sp[0], 0.86, 'indicated (r=0.85)', {color: '#ff9800'});
a2.text(sp[0], 0.71, 'inferred (r=0.70)', {color: '#f44336'});
a2.set_xlabel('drill hole spacing');
a2.set_ylabel('correlation');
a2.grid(true);
display(f2.show());

/// md
## slope of regression

Conditional bias. Slope = 1.0 is unbiased; < 1.0 means over-smoothing.

/// code
const slopes = results.map(r => r.slope);
const {fig: f3, ax: a3} = plot.subplots(1, 1, {figsize: [500, 200]});
a3.plot(sp, slopes, {color: '#c89b3c', linewidth: 2, marker: 'o'});
a3.axhline(1.0, {color: '#4caf50', linestyle: '--'});
a3.text(sp[0], 1.01, 'unbiased (slope=1)', {color: '#4caf50'});
a3.set_xlabel('drill hole spacing');
a3.set_ylabel('slope');
a3.grid(true);
display(f3.show());

---

**Workflow:** sgsim generates a reference model. At each spacing level, samples are placed on a regular grid and the field is re-estimated by kriging. The maps show spatial resolution degrading with wider spacing. The classification curve maps spacing to JORC/NI 43-101 confidence.
