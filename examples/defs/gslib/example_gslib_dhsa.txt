# drill hole spacing analysis

Determine optimal drill spacing for resource classification. Generate a reference model by unconditional simulation, decimate to progressively wider spacings, re-estimate at each level, and plot the spacing-vs-correlation classification curve.

/// code
const gridN = ui.slider("grid", 60, {min: 30, max: 100, step: 5});
const vRange = ui.slider("range", 15, {min: 5, max: 40, step: 1});
const vSill = ui.slider("sill", 1.0, {min: 0.2, max: 3.0, step: 0.1});
const vNugget = ui.slider("nugget", 0.1, {min: 0.0, max: 0.5, step: 0.05});
const simSeed = ui.slider("seed", 42, {min: 1, max: 999, step: 1});

/// code collapsed
// %collapsed
const { sgsim, kb2d } = await load("@atra/gslib");

const vario = { nugget: vNugget, structures: [{ type: "spherical", contribution: vSill, range: vRange }] };
const grid = { nx: gridN, ny: gridN, xsiz: 1, ysiz: 1 };
const searchRadius = Math.max(vRange * 2, 20);

const engine = sgsim({ grid, variogram: vario, search: { radius: searchRadius, ndmax: 12, nodmax: 12 } });
invalidation.then(() => engine.dispose());
const truth = engine.run(simSeed);
display(`reference: ${gridN}\u00d7${gridN}, range=${vRange}, sill=${vSill}, nugget=${vNugget}`);

/// md
## spacing analysis

/// code
const spacings = [2, 3, 4, 5, 6, 8, 10, 12, 15, 20, 25, 30].filter(s => s < gridN / 2);
const results = [];

for (const spacing of spacings) {
  const samples = [];
  for (let iy = 0; iy < gridN; iy += spacing)
    for (let ix = 0; ix < gridN; ix += spacing)
      samples.push([ix + 0.5, iy + 0.5, truth[iy * gridN + ix]]);
  if (samples.length < 4) continue;

  const est = kb2d({
    data: samples, grid, variogram: vario,
    search: { radius: Math.max(vRange * 2.5, spacing * 2), ndmax: Math.min(samples.length, 24) },
  }).est;

  let n = 0, st = 0, se = 0, st2 = 0, se2 = 0, ste = 0, serr2 = 0;
  for (let i = 0; i < gridN * gridN; i++) {
    if (est[i] <= -900) continue;
    const t = truth[i], e = est[i];
    n++; st += t; se += e; st2 += t * t; se2 += e * e; ste += t * e; serr2 += (e - t) ** 2;
  }
  if (n < 2) continue;

  const mt = st / n, me = se / n, vt = st2 / n - mt * mt, ve = se2 / n - me * me;
  const cov = ste / n - mt * me;
  results.push({
    spacing, nsamples: samples.length, est, samples,
    corr: vt > 0 && ve > 0 ? cov / Math.sqrt(vt * ve) : 0,
    slope: vt > 0 ? cov / vt : 0,
    rmse: Math.sqrt(serr2 / n),
  });
}

display(`\n${"spacing".padStart(7)}  ${"samples".padStart(7)}  ${"corr".padStart(7)}  ${"slope".padStart(7)}  ${"RMSE".padStart(7)}`);
display(`${"-------".padStart(7)}  ${"-------".padStart(7)}  ${"------".padStart(7)}  ${"------".padStart(7)}  ${"------".padStart(7)}`);
for (const r of results)
  display(`${String(r.spacing).padStart(7)}  ${String(r.nsamples).padStart(7)}  ${r.corr.toFixed(4).padStart(7)}  ${r.slope.toFixed(4).padStart(7)}  ${r.rmse.toFixed(4).padStart(7)}`);

/// md
## maps — truth vs selected spacings

/// code
// pick 4 evenly spaced results
const showIdx = results.length >= 4
  ? [0, Math.round((results.length - 1) / 3), Math.round(2 * (results.length - 1) / 3), results.length - 1]
  : results.map((_, i) => i);
const show = showIdx.map(i => results[i]);
const ncols = show.length + 1;
const sz = Math.min(180, Math.floor((window.innerWidth - 60) / ncols) - 10);
const cellW = sz / gridN;

// color range from truth
let vmin = Infinity, vmax = -Infinity;
for (let i = 0; i < truth.length; i++) { if (truth[i] < vmin) vmin = truth[i]; if (truth[i] > vmax) vmax = truth[i]; }
const vr = vmax - vmin || 1;

// truth
const ct = ui.canvas(sz, sz);
const ctxt = ct.getContext("2d");
for (let iy = 0; iy < gridN; iy++)
  for (let ix = 0; ix < gridN; ix++) {
    ctxt.fillStyle = std.viridis((truth[iy * gridN + ix] - vmin) / vr);
    ctxt.fillRect(ix * cellW, iy * cellW, cellW + 0.5, cellW + 0.5);
  }
display("truth");

// spacing maps
for (const r of show) {
  const ce = ui.canvas(sz, sz);
  const ctxe = ce.getContext("2d");
  for (let iy = 0; iy < gridN; iy++)
    for (let ix = 0; ix < gridN; ix++) {
      const v = r.est[iy * gridN + ix];
      ctxe.fillStyle = v < -900 ? "#222" : std.viridis((v - vmin) / vr);
      ctxe.fillRect(ix * cellW, iy * cellW, cellW + 0.5, cellW + 0.5);
    }
  for (const [sx, sy] of r.samples) {
    ctxe.beginPath(); ctxe.arc(sx / gridN * sz, sy / gridN * sz, 1.5, 0, Math.PI * 2);
    ctxe.fillStyle = "#fff"; ctxe.fill();
  }
  display(`spacing=${r.spacing}  r=${r.corr.toFixed(2)}`);
}

/// md
## classification curve

/// code
const w = Math.min(500, window.innerWidth - 80), h = 280;
const c = ui.canvas(w, h);
const ctx = c.getContext("2d");
ctx.fillStyle = "#1a1a1a"; ctx.fillRect(0, 0, w, h);

const pad = {l: 55, r: 20, t: 20, b: 40};
const pw = w - pad.l - pad.r, ph = h - pad.t - pad.b;

if (results.length > 0) {
  const maxSp = Math.max(...results.map(r => r.spacing)) * 1.1;
  const minCorr = Math.min(...results.map(r => r.corr));
  const yLo = Math.max(0, minCorr - 0.05);
  const sx = sp => pad.l + (sp / maxSp) * pw;
  const sy = corr => pad.t + ph - ((corr - yLo) / (1.0 - yLo)) * ph;

  ctx.strokeStyle = "#555"; ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(pad.l, pad.t); ctx.lineTo(pad.l, pad.t + ph); ctx.lineTo(pad.l + pw, pad.t + ph); ctx.stroke();

  for (const [thresh, color, label] of [[0.95, "#4caf50", "measured"], [0.85, "#ff9800", "indicated"], [0.70, "#f44336", "inferred"]]) {
    if (thresh < yLo) continue;
    const y = sy(thresh);
    ctx.setLineDash([6, 4]); ctx.strokeStyle = color; ctx.lineWidth = 1;
    ctx.beginPath(); ctx.moveTo(pad.l, y); ctx.lineTo(pad.l + pw, y); ctx.stroke();
    ctx.setLineDash([]); ctx.fillStyle = color; ctx.font = "10px monospace"; ctx.textAlign = "left";
    ctx.fillText(`${label} (r=${thresh})`, pad.l + 4, y - 3);
  }

  ctx.strokeStyle = "#c89b3c"; ctx.lineWidth = 2; ctx.beginPath();
  results.forEach((r, i) => { const x = sx(r.spacing), y = sy(r.corr); i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y); });
  ctx.stroke();

  for (const r of results) {
    ctx.beginPath(); ctx.arc(sx(r.spacing), sy(r.corr), 4, 0, Math.PI * 2);
    ctx.fillStyle = "#fff"; ctx.fill(); ctx.strokeStyle = "#000"; ctx.lineWidth = 1; ctx.stroke();
  }

  ctx.fillStyle = "#888"; ctx.font = "10px monospace"; ctx.textAlign = "center";
  ctx.fillText("drill hole spacing", pad.l + pw / 2, h - 5);
  for (const r of results) { ctx.fillStyle = "#666"; ctx.fillText(r.spacing, sx(r.spacing), pad.t + ph + 14); }
  ctx.textAlign = "right";
  for (const v of [0.7, 0.8, 0.9, 1.0]) { if (v >= yLo) { ctx.fillStyle = "#666"; ctx.fillText(v.toFixed(1), pad.l - 5, sy(v) + 3); } }
  ctx.save(); ctx.translate(12, pad.t + ph / 2); ctx.rotate(-Math.PI / 2); ctx.fillStyle = "#888"; ctx.textAlign = "center"; ctx.fillText("correlation", 0, 0); ctx.restore();
}

## slope of regression

/// code
const w2 = Math.min(500, window.innerWidth - 80), h2 = 200;
const c2 = ui.canvas(w2, h2);
const ctx2 = c2.getContext("2d");
ctx2.fillStyle = "#1a1a1a"; ctx2.fillRect(0, 0, w2, h2);

const pad2 = {l: 55, r: 20, t: 15, b: 40};
const pw2 = w2 - pad2.l - pad2.r, ph2 = h2 - pad2.t - pad2.b;

if (results.length > 0) {
  const maxSp = Math.max(...results.map(r => r.spacing)) * 1.1;
  const sx2 = sp => pad2.l + (sp / maxSp) * pw2;
  const sy2 = sl => pad2.t + ph2 - (sl / 1.2) * ph2;

  ctx2.strokeStyle = "#555"; ctx2.lineWidth = 1;
  ctx2.beginPath(); ctx2.moveTo(pad2.l, pad2.t); ctx2.lineTo(pad2.l, pad2.t + ph2); ctx2.lineTo(pad2.l + pw2, pad2.t + ph2); ctx2.stroke();

  ctx2.setLineDash([6, 4]); ctx2.strokeStyle = "#4caf50";
  ctx2.beginPath(); ctx2.moveTo(pad2.l, sy2(1.0)); ctx2.lineTo(pad2.l + pw2, sy2(1.0)); ctx2.stroke();
  ctx2.setLineDash([]); ctx2.fillStyle = "#4caf50"; ctx2.font = "10px monospace"; ctx2.textAlign = "left";
  ctx2.fillText("unbiased (slope=1)", pad2.l + 4, sy2(1.0) - 3);

  ctx2.strokeStyle = "#c89b3c"; ctx2.lineWidth = 2; ctx2.beginPath();
  results.forEach((r, i) => { const x = sx2(r.spacing), y = sy2(r.slope); i === 0 ? ctx2.moveTo(x, y) : ctx2.lineTo(x, y); });
  ctx2.stroke();

  for (const r of results) {
    ctx2.beginPath(); ctx2.arc(sx2(r.spacing), sy2(r.slope), 4, 0, Math.PI * 2);
    ctx2.fillStyle = "#fff"; ctx2.fill(); ctx2.strokeStyle = "#000"; ctx2.lineWidth = 1; ctx2.stroke();
  }

  ctx2.fillStyle = "#888"; ctx2.font = "10px monospace"; ctx2.textAlign = "center";
  ctx2.fillText("drill hole spacing", pad2.l + pw2 / 2, h2 - 5);
  ctx2.save(); ctx2.translate(12, pad2.t + ph2 / 2); ctx2.rotate(-Math.PI / 2); ctx2.textAlign = "center"; ctx2.fillText("slope", 0, 0); ctx2.restore();
}

**Workflow:** sgsim generates a reference model. At each spacing level, samples are placed on a regular grid and the field is re-estimated by kriging. The maps show spatial resolution degrading with wider spacing. The classification curve maps spacing to JORC/NI 43-101 confidence.
