# gsjs — krige once, re-realize for free

[**@gcu/gsjs**](ext/gsjs/README.md) is the modern geostatistics library — the evolving sibling to the frozen `@gcu/gslib` oracle. Its trick: kriging captures a **BlockEstimateTensor** (the weights + sample indices per block), so the estimate is *linear in the sample values*. That means a **top-cut** (capping high grades) is a free re-weighting — no re-kriging.

Drag the **top-cut** slider below. The kriging solve ran **once**; every slider move only re-runs `evaluate()` — a sparse matrix-vector product over weights already in hand. Watch the re-realize time vs the krige time.

/// code collapsed
// %collapsed
// synthetic deposit: a few Gaussian anomalies + background, sampled at random
// "drillhole" locations with multiplicative noise (so the upper tail is skewed —
// the high-grade outliers a top-cut is meant to tame).
function mulberry32(a){return function(){a|=0;a=a+0x6D2B79F5|0;let t=Math.imul(a^a>>>15,1|a);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};}
const rng = mulberry32(7);
const EXTENT = 500;
const anomalies = [[150,150,6,70],[350,300,9,55],[260,410,4,90],[410,120,7,50]];
function trueGrade(x,y){
  let g = 0.4;
  for (const [ax,ay,pk,r] of anomalies){ const d2=(x-ax)**2+(y-ay)**2; g += pk*Math.exp(-d2/(2*r*r)); }
  return g;
}
const N = 70;
const rows = [];
for (let i=0;i<N;i++){
  const x = rng()*EXTENT, y = rng()*EXTENT;
  const grade = Math.max(0, trueGrade(x,y) * (0.6 + 0.9*rng()));  // skewed → real outliers
  rows.push({ X:x, Y:y, Z:0, AU:grade });
}

/// code
// the recipe: a single-domain ordinary-kriging estimation, 50×50 blocks over the
// 500×500 m extent. `realization: topcut` declares the transform; its `cap` is
// what the slider drives. (gsjs loaded once; estimate() is the expensive stage.)
const gsjs = await load("@gcu/gsjs");
const grid = { nx:50, ny:50, nz:1, xmn:5, ymn:5, zmn:0, xsiz:10, ysiz:10, zsiz:10 };

const r = gsjs.recipe({
  data: { columns: { x:"X", y:"Y", z:"Z", value:"AU" } },
  block_grid: { ...grid, discretization:[2,2,1] },
  default_model: gsjs.ok({
    variogram: gsjs.variogram([{ type:"spherical", cc:1.0, aa:130 }], { c0:0.15 }),
    search: gsjs.search({ radius:220, ndmin:2, ndmax:24 }),
    realization: gsjs.topcut({ cap:99 }),
  }),
  output: { aggregations: [{ kind:"gradeTonnage", cutoffs:[0.5,1,1.5,2,3,4,5] }] },
});

const t0 = performance.now();
const kriged = gsjs.estimate(r, { rows });   // ← search + solve, ONCE
const tKrige = performance.now() - t0;

/// code
// the interactive surface. the top-cut slider's onInput re-runs ONLY evaluate()
// (re-realize) — the DAG never re-executes, the kriging system is never re-solved.
const size = Math.min(480, window.innerWidth - 80);
const cw = size / grid.nx;
const map = ui.canvas(size, size);
const mctx = map.getContext("2d");
const curve = ui.canvas(size, 150);
const gctx = curve.getContext("2d");
const readout = document.createElement("div");
readout.style.cssText = "font-family:monospace;font-size:12px;margin-top:6px;line-height:1.5";
ui.display(readout);

// fixed color range from the uncapped field, so capping visibly cools the hotspots
const uncapped = gsjs.evaluate(kriged, {});
let vmax = 0;
for (let i=0;i<uncapped.estimates.length;i++) if (uncapped.status[i]===gsjs.STATUS.OK) vmax = Math.max(vmax, uncapped.estimates[i]);

function drawMap(est, status){
  for (let iy=0;iy<grid.ny;iy++) for (let ix=0;ix<grid.nx;ix++){
    const idx = iy*grid.nx+ix;
    mctx.fillStyle = status[idx]!==gsjs.STATUS.OK ? "#1a1a1a" : std.viridis(Math.min(1, est[idx]/vmax));
    mctx.fillRect(ix*cw, (grid.ny-1-iy)*cw, cw+0.5, cw+0.5);  // flip y → north up
  }
  for (const s of rows){
    mctx.beginPath();
    mctx.arc(s.X/EXTENT*size, (1-s.Y/EXTENT)*size, 2.5, 0, 7);
    mctx.fillStyle = "#fff"; mctx.fill();
    mctx.strokeStyle = "#000"; mctx.lineWidth = 1; mctx.stroke();
  }
}

function drawCurve(gt){
  gctx.fillStyle = "#141414"; gctx.fillRect(0,0,size,150);
  const cuts = gt.cutoffs, grade = gt.grade, ton = gt.tonnage;
  const maxCut = cuts[cuts.length-1], maxGrade = vmax, maxTon = ton[0] || 1;
  const px = (c)=> 30 + (c/maxCut)*(size-45);
  const pyG = (g)=> 140 - (g/maxGrade)*125;
  const pyT = (t)=> 140 - (t/maxTon)*125;
  // tonnage (area, muted) then mean-grade-above-cutoff (amber line)
  gctx.strokeStyle = "#3a6ea5"; gctx.lineWidth = 1.5; gctx.beginPath();
  for (let i=0;i<cuts.length;i++){ const X=px(cuts[i]),Y=pyT(ton[i]); i?gctx.lineTo(X,Y):gctx.moveTo(X,Y); }
  gctx.stroke();
  gctx.strokeStyle = "#c89b3c"; gctx.lineWidth = 2; gctx.beginPath();
  for (let i=0;i<cuts.length;i++){ const X=px(cuts[i]),Y=pyG(grade[i]); i?gctx.lineTo(X,Y):gctx.moveTo(X,Y); }
  gctx.stroke();
  gctx.fillStyle = "#888"; gctx.font = "11px monospace";
  gctx.fillText("grade (amber) · tonnage (blue) vs cut-off", 32, 14);
}

function update(cap){
  const t1 = performance.now();
  const { estimates, status, aggregations } = gsjs.evaluate(kriged, { _default:{ transform:"topcut", transform_params:{ cap } } });
  const tEval = performance.now() - t1;
  drawMap(estimates, status);
  drawCurve(aggregations.gradeTonnage);
  let sum=0, n=0; for (let i=0;i<estimates.length;i++) if (status[i]===gsjs.STATUS.OK){ sum+=estimates[i]; n++; }
  readout.innerHTML =
    `top-cut <b>${cap.toFixed(1)}</b> g/t &nbsp;·&nbsp; mean estimate <b>${(sum/n).toFixed(3)}</b> g/t<br>` +
    `re-realize <b>${tEval.toFixed(2)} ms</b> &nbsp;·&nbsp; (kriging solve was <b>${tKrige.toFixed(0)} ms</b>, run once) &nbsp;·&nbsp; ${n} blocks`;
}

ui.slider("top-cut (g/t)", Math.round(vmax*0.6), { min:1, max:Math.ceil(vmax), step:0.5, onInput:update });
update(Math.round(vmax*0.6));

/// md
**Why this works.** Kriging is a weighted average: `estimate = Σ wᵢ·zᵢ`. `gsjs.estimate()` solves the kriging system once and hands back the **weights and sample indices** for every block (the *BlockEstimateTensor*). A top-cut replaces each `zᵢ` with `min(zᵢ, cap)` — still a weighted sum of the *same* weights, so `gsjs.evaluate()` just recombines them. The expensive search + solve never repeats.

This is the staged split the recipe API is built around: `estimate()` is the slow, rarely-changing stage; `evaluate()` is the cheap tail a live slider re-runs. The same property makes **high-grade restriction** (HGR, distance-aware capping) a free re-realize too — see the [full-tour example](example_gsjs_tour.html). For domained estimation (separate models per geological zone), see the [multi-domain example](example_gsjs_domains.html).
