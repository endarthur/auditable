# atra multi-memory — columnar block model

WebAssembly's single linear memory caps at 4 GB (i32 addressing). large datasets like block models (millions of blocks with many attributes) can hit this wall. the **multi-memory** proposal (Chrome 120+, Firefox 125+, Edge 120+) provides multiple independent memory banks, each with its own 4 GB address space and zero performance penalty.

this notebook uses separate memory banks for spatial coordinates, grade values, and scratch space — a natural columnar layout for geostatistical data.

/// code
const { atra } = await load("./ext/atra/index.js");

/// code
const N = ui.slider("blocks", 5000, {min: 1000, max: 50000, step: 1000});
const cutoff = ui.slider("cutoff", 0.5, {min: 0.0, max: 2.0, step: 0.01});

/// code
// check multi-memory support before proceeding
if (!atra.hasMultiMemory) {
  ui.display("multi-memory not supported in this browser (requires Chrome 120+, Firefox 125+, or Edge 120+)");
  throw new Error("multi-memory unsupported");
}

// three independent memory banks — each can grow to 4 GB independently
const spatial = new WebAssembly.Memory({ initial: 4 });   // x, y, z coords (f64)
const grades  = new WebAssembly.Memory({ initial: 4 });   // grade values (f64)
const scratch = new WebAssembly.Memory({ initial: 1 });   // mask indices (i32)

const wasm = atra({ memory: { spatial, grades, scratch } })`
  memory spatial
  memory grades
  memory scratch

  ! count blocks where grade exceeds cutoff, store their indices in mask
  function grade_above_cutoff(
      g: grades array f64;
      mask: scratch array i32;
      n: i32;
      cutoff: f64
  ): i32
  var i, count: i32
  begin
    count := 0
    for i := 0, n
      if (g[i] >= cutoff) then
        mask[count] := i
        count := count + 1
      end if
    end for
    grade_above_cutoff := count
  end

  ! compute mean grade of selected blocks
  function mean_grade(
      g: grades array f64;
      mask: scratch array i32;
      count: i32
  ): f64
  var i, idx: i32, total: f64
  begin
    total := 0.0
    for i := 0, count
      idx := mask[i]
      total := total + g[idx]
    end for
    mean_grade := total / f64(count)
  end

  ! compute distance from origin for each block, store in scratch f64 array
  subroutine compute_distances(
      x: spatial array f64;
      y: spatial array f64;
      dist: scratch array f64;
      n: i32
  )
  var i: i32, dx, dy: f64
  begin
    for i := 0, n
      dx := x[i]
      dy := y[i]
      dist[i] := sqrt(dx * dx + dy * dy)
    end for
  end

  ! spatial query: count blocks within a radius from origin
  function blocks_in_radius(
      x: spatial array f64;
      y: spatial array f64;
      n: i32;
      radius: f64
  ): i32
  var i, count: i32, dx, dy, r2, rlim: f64
  begin
    rlim := radius * radius
    count := 0
    for i := 0, n
      dx := x[i]
      dy := y[i]
      r2 := dx * dx + dy * dy
      if (r2 <= rlim) then
        count := count + 1
      end if
    end for
    blocks_in_radius := count
  end
`;

ui.display(`compiled: ${Object.keys(wasm).filter(k => !k.startsWith('_')).join(', ')}`);

/// code
// %manual
// populate memory banks with synthetic block model data
function mulberry32(a) {
  return function() {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}
const rng = mulberry32(42);

// grow memories if needed
const spatialBytes = N * 2 * 8; // x, y as f64
const gradeBytes = N * 8;       // one grade per block
const scratchBytes = N * 8;     // mask (i32) or distances (f64)
function ensurePages(mem, bytes) {
  const needed = Math.ceil(bytes / 65536);
  const current = mem.buffer.byteLength / 65536;
  if (current < needed) mem.grow(needed - current);
}
ensurePages(spatial, spatialBytes);
ensurePages(grades, gradeBytes);
ensurePages(scratch, scratchBytes);

// spatial bank: x coords at offset 0, y coords at offset N*8
const xArr = new Float64Array(spatial.buffer, 0, N);
const yArr = new Float64Array(spatial.buffer, N * 8, N);
for (let i = 0; i < N; i++) {
  xArr[i] = (rng() - 0.5) * 200;  // -100 to +100
  yArr[i] = (rng() - 0.5) * 200;
}

// grades bank: one f64 per block (lognormal-ish distribution)
const gArr = new Float64Array(grades.buffer, 0, N);
for (let i = 0; i < N; i++) {
  // distance-weighted grade: higher near center
  const r = Math.sqrt(xArr[i] ** 2 + yArr[i] ** 2);
  gArr[i] = Math.max(0, 2.0 * Math.exp(-r / 40) + (rng() - 0.5) * 0.8);
}

// run cutoff query
const xPtr = 0, yPtr = N * 8, gPtr = 0, maskPtr = 0;
const nAbove = wasm.grade_above_cutoff(gPtr, maskPtr, N, cutoff);
const meanG = nAbove > 0 ? wasm.mean_grade(gPtr, maskPtr, nAbove) : 0;
const inRadius = wasm.blocks_in_radius(xPtr, yPtr, N, 50.0);

ui.display(`${N} blocks | cutoff \u2265 ${cutoff.toFixed(2)}: ${nAbove} blocks (${(100*nAbove/N).toFixed(1)}%) | mean grade above cutoff: ${meanG.toFixed(3)}`);
ui.display(`blocks within radius 50: ${inRadius} (${(100*inRadius/N).toFixed(1)}%)`);

// visualize: block map colored by grade, highlight above-cutoff
const W = Math.min(560, window.innerWidth - 80), H = W;
const c = ui.canvas(W, H);
const ctx = c.getContext("2d");

ctx.fillStyle = "#0a0a0a";
ctx.fillRect(0, 0, W, H);

// read mask indices
const maskArr = new Int32Array(scratch.buffer, 0, nAbove);
const aboveSet = new Set(maskArr);

// scale: map [-100, 100] to [20, W-20]
const scale = (v) => (v + 100) / 200 * (W - 40) + 20;

// draw all blocks (dim)
for (let i = 0; i < N; i++) {
  const x = scale(xArr[i]);
  const y = scale(yArr[i]);
  const g = gArr[i];
  if (aboveSet.has(i)) continue; // draw highlighted ones on top
  const lum = 15 + Math.min(g, 2) * 10;
  ctx.fillStyle = `hsl(210, 30%, ${lum}%)`;
  ctx.fillRect(x - 1, y - 1, 2, 2);
}

// draw above-cutoff blocks (bright)
for (let i = 0; i < nAbove; i++) {
  const idx = maskArr[i];
  const x = scale(xArr[idx]);
  const y = scale(yArr[idx]);
  const g = gArr[idx];
  const hue = 45 - Math.min(g, 2) * 15; // gold to orange
  const lum = 45 + Math.min(g, 2) * 15;
  ctx.fillStyle = `hsl(${hue}, 90%, ${lum}%)`;
  ctx.fillRect(x - 1.5, y - 1.5, 3, 3);
}

// draw radius circle
ctx.strokeStyle = "rgba(200, 155, 60, 0.4)";
ctx.lineWidth = 1;
ctx.beginPath();
const r50px = 50 / 200 * (W - 40);
ctx.arc(W / 2, H / 2, r50px, 0, Math.PI * 2);
ctx.stroke();

// memory usage summary
const spatialKB = (spatial.buffer.byteLength / 1024).toFixed(0);
const gradesKB = (grades.buffer.byteLength / 1024).toFixed(0);
const scratchKB = (scratch.buffer.byteLength / 1024).toFixed(0);
ui.display(`memory banks: spatial=${spatialKB} KB, grades=${gradesKB} KB, scratch=${scratchKB} KB`);

/// md
**how it works:** three `memory` declarations create separate Wasm memory banks. parameters like `g: grades array f64` bind arrays to specific banks — the compiler emits load/store instructions with the correct memory index. each bank grows independently, so a 50M-block model could use ~1.2 GB for coordinates, ~3 GB for grades, and scratch space as needed — all within i32 addressing, no Memory64 required. `atra.hasMultiMemory` lets you check support before compiling.
